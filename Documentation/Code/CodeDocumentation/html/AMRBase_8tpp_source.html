64 /**
65  * @brief Destroy the AMRBase object, removing its pilot variables from the
66  * registry.
67  *
68  * @tparam VAR Variable container type this AMR strategy operates on.
69  */
70 template <class VAR>
71 AMRBase<VAR>::~AMRBase() {
72  auto& registry = PilotRegistry();
73  auto it_mesh = registry.find(&this->par_mesh_);
74  if (it_mesh == registry.end()) {
75  return;
76  }
77  auto& owners = it_mesh->second;
78  for (auto it = owners.begin(); it != owners.end();) {
79  it = (it->second == this) ? owners.erase(it) : std::next(it);
80  }
81  if (owners.empty()) {
82  registry.erase(it_mesh);
83  }
84 }

142 /**
143  * @brief Accumulate (max) into `combined_error` the local errors of `var`,
144  * estimated with the estimator of this AMR object and normalized by
145  * its derefinement threshold.
146  *
147  * @details Each pilot variable is thus judged with the criterion of the AMR
148  * object it drives (estimator, `max_elem_error`,
149  * `scale_down_factor`): a normalized error lower than 1 means
150  * "derefinement allowed" for this variable.
151  *
152  * @tparam VAR Variable container type this AMR strategy operates on.
153  * @param var Pilot variable of this AMR object.
154  * @param combined_error Per-element normalized error, updated in place.
155  */
156 template <class VAR>
157 void AMRBase<VAR>::AccumulateNormalizedError(Variable_t& var, mfem::Vector& combined_error) {
158  this->EnsureCriteriaSet();
159  const double threshold = this->amr_scale_down_factor_ * this->amr_max_elem_error_;
160  MFEM_VERIFY(threshold > 0., "AMRBase: derefinement threshold must be strictly positive.");
161  auto estimator =
162  this->amr_estimator_->get_value(var.get_ref_gf(), *var.get_fespace(), this->par_mesh_);
163  const mfem::Vector& local_errors = estimator->GetLocalErrors();
164  const int ne = combined_error.Size();
165  MFEM_VERIFY(local_errors.Size() == ne, "AMRBase: error vector size mismatch.");
166  const double inv_threshold = 1. / threshold;
167  for (int e = 0; e < ne; e++) {
168  combined_error(e) = std::max(combined_error(e), local_errors(e) * inv_threshold);
169  }
170 }

172 /**
173  * @brief Derefine the mesh once, only where every pilot variable concerned
174  * agrees.
175  *
176  * @details The variables concerned are:
177  * - the pilot variables of this AMR object;
178  * - the auxiliary variables of the owning Problem (`coupled_vars`)
179  * that are pilot variables of an AMR object on the same mesh
180  * (i.e. associated with a refinement criterion in their own
181  * Problem). Each of them is judged with the criterion of that
182  * AMR object.
183  * The other variables (auxiliary variables without refinement
184  * criterion, variables of non coupled Problems, MPI rank field...)
185  * are ignored. The per-element normalized errors are combined via
186  * `max()`, then `mfem::ParMesh::DerefineByError()` is applied once
187  * with a unit threshold (sum of the sibling errors, as
188  * `mfem::ThresholdDerefiner`): for a single variable, this is
189  * exactly the derefinement with the threshold
190  * `scale_down_factor * max_elem_error`.
191  *
192  * The variables are visited in the order of `vars` then
193  * `coupled_vars`, identical on every rank (the estimators may
194  * exchange data between ranks).
195  *
196  * @tparam VAR Variable container type this AMR strategy operates on.
197  * @param vars Variables of the Problem owning this AMR object.
198  * @param coupled_vars Auxiliary variables of the owning Problem.
199  * @return true if the mesh was derefined.
200  */
201 template <class VAR>
202 bool AMRBase<VAR>::DerefineByConsensus(VAR& vars, const std::vector<VAR*>& coupled_vars) {
203  this->RegisterPilots(vars);
204  const int ne = this->par_mesh_.GetNE();
205  mfem::Vector combined_error(ne);
206  combined_error = 0.0;
207 
208  std::vector<const void*> done;
209  auto add = [&done, &combined_error](Variable_t& var, AMRBase<VAR>* owner) {
210  const void* key = static_cast<const void*>(&var);
211  if (std::find(done.begin(), done.end(), key) != done.end()) {
212  return;
213  }
214  done.push_back(key);
215  owner->AccumulateNormalizedError(var, combined_error);
216  };
217 
218  for (auto* pilot : this->PilotVariables(vars)) {
219  add(*pilot, this);
220  }
221  const auto& owners = PilotRegistry()[&this->par_mesh_];
222  for (auto* container : coupled_vars) {
223  if (container == nullptr) {
224  continue;
225  }
226  for (std::size_t i = 0; i < container->get_variables_number(); i++) {
227  auto& var = (*container)[i];
228  if (!this->IsOnMesh(var.get_fespace())) {
229  continue;
230  }
231  auto it = owners.find(static_cast<const void*>(&var));
232  if (it != owners.end()) {
233  add(var, it->second);
234  }
235  }
236  }
237 
238  return this->par_mesh_.DerefineByError(combined_error, 1.0, this->amr_nc_limit_);
239 }

243  * by this AMR strategy.
244  *
245  *
246  * @tparam VAR Variable container type this AMR strategy operates on.
247  * @param amr_params Collection of Parameter objects for AMR :
248  * - max_elem_error (mandatory): Local error threshold above which an element is
249  * marked for refinement (used directly for

251  * derefinement).

361 /**
362  * @brief Resynchronize every variable sharing the mesh with its current
363  * state, after one mesh modification (refinement, derefinement or
364  * rebalancing).
365  *
366  * @details Variables are gathered from `vars` and `auxvars` without
367  * duplicates (the variables of the Problem owning this AMR object
368  * are usually also part of `auxvars`), keeping only those defined
369  * on the mesh managed by this object, and several variables may
370  * share the same finite element space (same SpatialDiscretization).
371  * The update is thus done in four stages, so that each space is
372  * updated once and its transfer operator stays alive until every
373  * grid function defined on it has been transferred:
374  * 1. snapshot of the multi-timestep history of each variable
375  * (`Variable::prepareMeshTransfer()`), before any space update;
376  * 2. `Update()` of each distinct finite element space;
377  * 3. transfer of each grid function and of its history, with the
378  * same transfer operator (interpolation or rebalancing matrix);
379  * 4. `UpdatesFinished()` of each distinct finite element space.
380  *
381  * @tparam VAR Variable container type this AMR strategy operates on.
382  * @param vars Primary variables sharing the mesh managed by this AMR object.
383  * @param auxvars Auxiliary variable collections sharing the same mesh.
384  */
385 template <class VAR>
386 void AMRBase<VAR>::UpdateVariables(VAR& vars, const std::vector<VAR*>& auxvars) {
387  using Var = std::remove_reference_t<decltype(vars[0])>;
388 
389  std::vector<Var*> unique_vars;
390  auto collect = [this, &unique_vars](VAR& container) {
391  for (std::size_t i = 0; i < container.get_variables_number(); i++) {
392  Var* var = &container[i];
393  // Variables defined on another mesh are not affected by this mesh modification
394  if (!this->IsOnMesh(var->get_fespace())) {
395  continue;
396  }
397  if (std::find(unique_vars.begin(), unique_vars.end(), var) == unique_vars.end()) {
398  unique_vars.push_back(var);
399  }
400  }
401  };
402  collect(vars);
403  for (auto* auxvar : auxvars) {
404  if (auxvar != nullptr) {
405  collect(*auxvar);
406  }
407  }
408 
409  // 1. Snapshots, before any space update
410  std::vector<mfem::ParFiniteElementSpace*> fespaces;
411  std::vector<std::map<int, mfem::ParGridFunction>> histories(unique_vars.size());
412  for (std::size_t k = 0; k < unique_vars.size(); k++) {
413  unique_vars[k]->prepareMeshTransfer(histories[k]);
414  auto* fes = unique_vars[k]->get_fespace();
415  if (std::find(fespaces.begin(), fespaces.end(), fes) == fespaces.end()) {
416  fespaces.push_back(fes);
417  }
418  }
419  // 2. One update per space
420  for (auto* fes : fespaces) {
421  fes->Update();
422  }
423  // 3. Transfer of the solution and of its history with the same operator
424  for (std::size_t k = 0; k < unique_vars.size(); k++) {
425  auto* var = unique_vars[k];
426  auto& gf = var->get_ref_gf();
427  gf.Update();
428  var->finalizeMeshTransfer(histories[k]);
429  mfem::Vector unk;
430  gf.GetTrueDofs(unk);
431  var->updateAfterMeshChange(unk);
432  }
433  // 4. Free the transfer operators
434  for (auto* fes : fespaces) {
435  fes->UpdatesFinished();
436  }
437 }

439 /**
440  * @brief Rebalance the parallel mesh among the MPI ranks, then
441  * resynchronize every variable sharing it.
442  *
443  * @details No-op if rebalancing is disabled (`rebalance` parameter), if
444  * the computation runs on a single rank, or if the mesh is
445  * conforming (`mfem::ParMesh::Rebalance()` is only available for
446  * nonconforming meshes). Rebalancing is a mesh modification of
447  * its own: it must be followed by a dedicated update of the
448  * spaces and grid functions (it cannot be merged with the update
449  * following a refinement). The estimator flux spaces do not need
450  * any update here: the estimator updates them itself.
451  *
452  * @tparam VAR Variable container type this AMR strategy operates on.
453  * @param vars Primary variables sharing the mesh managed by this AMR object.
454  * @param auxvars Auxiliary variable collections sharing the same mesh.
455  * @return true if the mesh was rebalanced.
456  */
457 template <class VAR>
458 bool AMRBase<VAR>::Rebalance(VAR& vars, const std::vector<VAR*>& auxvars) {
459  if (!this->amr_rebalance_ || this->par_mesh_.GetNRanks() == 1) {
460  return false;
461  }
462  if (!this->par_mesh_.Nonconforming()) {
463  SlothInfo::verbose("AMRBase::Rebalance: conforming mesh, rebalancing skipped.");
464  return false;
465  }
466  this->par_mesh_.Rebalance();
467  this->UpdateVariables(vars, auxvars);
468  // The estimator flux spaces are updated by the estimator itself at the next AMR pass.
469  return true;
470 }

472 /**
473  * @brief Print (on rank 0, Verbose mode) the distribution of the mesh
474  * elements among the MPI ranks: min, max and mean number of
475  * elements per rank, and imbalance factor max/mean.
476  *
477  * @tparam VAR Variable container type this AMR strategy operates on.
478  * @param stage Label of the AMR stage that has just been applied.
479  */
480 template <class VAR>
481 void AMRBase<VAR>::PrintLoadBalance(const std::string& stage) const {
482  MPI_Comm comm = this->par_mesh_.GetComm();
483  const int nranks = this->par_mesh_.GetNRanks();
484  const long long local_ne = this->par_mesh_.GetNE();
485  long long min_ne = 0;
486  long long max_ne = 0;
487  long long sum_ne = 0;
488  MPI_Allreduce(&local_ne, &min_ne, 1, MPI_LONG_LONG, MPI_MIN, comm);
489  MPI_Allreduce(&local_ne, &max_ne, 1, MPI_LONG_LONG, MPI_MAX, comm);
490  MPI_Allreduce(&local_ne, &sum_ne, 1, MPI_LONG_LONG, MPI_SUM, comm);
491  if (this->par_mesh_.GetMyRank() == 0) {
492  const double mean_ne = static_cast<double>(sum_ne) / nranks;
493  const double imbalance = mean_ne > 0. ? static_cast<double>(max_ne) / mean_ne : 1.;
494  SlothInfo::verbose("AMR load balance (", stage, "): ", sum_ne, " elements on ", nranks,
495  " ranks, min/max/mean = ", min_ne, "/", max_ne, "/", mean_ne,
496  ", imbalance (max/mean) = ", imbalance);
497  }
498 }

500 /**
501  * @brief Apply the initial refinement loop on the initial condition,
502  * before the time-stepping loop starts.
503  *
504  * @details After each mesh refinement cycle (up to `amr_max_preref_cycles_`):
505  * - `Refine()` decides whether to refine, based on `vars`;
506  * - every variable of `vars` and `auxvars` is resynchronized
507  * (`UpdateVariables()`) — all of them share the same mesh and
508  * must stay consistent with its current state;
509  * - the mesh is rebalanced if requested (`Rebalance()`), so that
510  * the next cycle starts from a balanced partition;
511  * - each variable is then reset to its initial condition via
512  * `Variable::setInitialCondition()`.
513  *
514  * @tparam VAR Variables container type this AMR strategy operates on.
515  * @param vars Primary variables sharing the mesh managed by this AMR
516  * object.
517  * @param auxvars Auxiliary variable collections sharing the same mesh,
518  * if any (empty by default).
519  */
520 template <class VAR>
521 void AMRBase<VAR>::InitialRefine(VAR& vars, std::vector<VAR*> auxvars) {
522  this->RegisterPilots(vars);
523  int cycle = 0;
524  bool did_refine = false;
525  while (cycle < this->amr_max_preref_cycles_) {
526  did_refine = this->Refine(vars);
527 
528  if (!did_refine) {
529  break;
530  }
531 
532  this->UpdateVariables(vars, auxvars);
533  const bool did_rebalance = this->Rebalance(vars, auxvars);
534 
535  for (std::size_t i = 0; i < vars.get_variables_number(); i++) {
536  vars[i].setInitialCondition();
537  }
538  for (auto* auxvar_container : auxvars) {
539  for (std::size_t i = 0; i < auxvar_container->get_variables_number(); i++) {
540  (*auxvar_container)[i].setInitialCondition();
541  }
542  }
543 
544  cycle++;
545  this->PrintLoadBalance("pre-refinement cycle " + std::to_string(cycle) +
546  (did_rebalance ? ", rebalanced" : ""));
547  }
548  SlothInfo::verbose("Pre-refinement: ", cycle, " cycles applied on initial condition.");
549 }

577  * @brief Apply one derefinement pass during the time-stepping loop.
578  *
579  * @details - `Derefine()` decides based on the pilot variables of this
580  * object and on the auxiliary variables of the owning Problem
581  * that are pilot variables of another AMR object on the same
582  * mesh (see `DerefineByConsensus()`), not on every variable of
583  * `auxvars`;
584  * - if the mesh was derefined, every variable in `vars` AND
585  * `auxvars` is resynchronized via `UpdateVariables()`, and a
586  * rebalancing is scheduled (see `StepRebalance()`).
587  *
588  * @tparam VAR Variable container type this AMR strategy operates on.
589  * @param vars Collection of variables sharing the mesh managed by this
590  * AMR object.
591  * @param auxvars Variable collections of every Problem of the coupling,
592  * resynchronized after a derefinement (empty by default).
593  */
594 template <class VAR>
595 void AMRBase<VAR>::StepDerefine(VAR& vars, std::vector<VAR*> auxvars) {
596  bool did_derefine = this->Derefine(vars, this->coupled_vars_);
597 
598  if (did_derefine) {
599  this->UpdateVariables(vars, auxvars);
600  this->pending_rebalance_ = true;
601  }
602 }

604 /**
605  * @brief Rebalance the mesh once at the end of an AMR pass, if the mesh
606  * was modified by `StepDerefine()` and/or `StepRefine()` since the
607  * last call.
608  *
609  * @details A single rebalancing per pass (derefine + refine) avoids two
610  * redistributions of the mesh per time step. The load balance is
611  * printed after each pass that modified the mesh.
612  *
613  * @tparam VAR Variable container type this AMR strategy operates on.
614  * @param vars Primary variables sharing the mesh managed by this AMR object.
615  * @param auxvars Auxiliary variable collections sharing the same mesh.
616  */
617 template <class VAR>
618 void AMRBase<VAR>::StepRebalance(VAR& vars, std::vector<VAR*> auxvars) {
619  if (!this->pending_rebalance_) {
620  return;
621  }
622  this->pending_rebalance_ = false;
623  const bool did_rebalance = this->Rebalance(vars, auxvars);
624  this->PrintLoadBalance(did_rebalance ? "AMR pass, rebalanced" : "AMR pass");
625 }

627 /**
628  * @brief Filter refinement candidates that would exceed the configured
629  * maximum refinement depth for their element.
630  *
631  * @details No-op if amr_max_level_ is 0 (no limit configured, default).
632  * Prevents an element from being refined indefinitely across many
633  * time steps: each StepRefine() call only refines once, so
634  * without this filter an element could accumulate one extra
635  * refinement level per time step with no upper bound.
636  *
637  * @tparam VAR Variable container type this AMR strategy operates on.
638  * @param candidates Refinement candidates before filtering.
639  * @return mfem::Array<mfem::Refinement> Filtered candidates, keeping only
640  * elements whose current depth is below amr_max_level_.
641  */
642 template <class VAR>
643 mfem::Array<mfem::Refinement> AMRBase<VAR>::FilterRefinedCandidates(
644  const mfem::Array<mfem::Refinement>& candidates) {
645  if (this->amr_max_level_ <= 0) {
646  return candidates;
647  }
648 
649  MFEM_VERIFY(this->par_mesh_.ncmesh != nullptr,
650  "FilterRefinedCandidates requires a nonconforming mesh (ncmesh == nullptr).");
651 
652  mfem::Array<mfem::Refinement> filtered_candidates;
653  for (int i = 0; i < candidates.Size(); i++) {
654  int elem = candidates[i].index;
655  int depth = this->par_mesh_.ncmesh->GetElementDepth(elem);
656  if (depth < this->amr_max_level_) {
657  filtered_candidates.Append(candidates[i]);
658  }
659  }
660  return filtered_candidates;
661 }
