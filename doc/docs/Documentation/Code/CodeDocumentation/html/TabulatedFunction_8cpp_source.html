84 std::vector<TabulatedFunction::Sample> TabulatedFunction::read_csv(
85  const std::filesystem::path& path) {
86  std::ifstream file(path);
87  MFEM_VERIFY(file.is_open(), "TabulatedFunction: cannot open CSV file '" << path.string() << "'.");
88 
89  std::vector<Sample> samples;
90  std::string line;
91  // Read header
92  std::getline(file, line);
93 
94  int expectedColumns = -1;
95  int lineNo = 1;
96 
97  while (std::getline(file, line)) {
98  ++lineNo;
99  if (line.empty()) continue;
100 
101  std::vector<std::string> fields;
102  std::stringstream ss(line);
103  std::string token;
104  while (std::getline(ss, token, ',')) fields.push_back(token);
105 
106  if (expectedColumns < 0) {
107  expectedColumns = static_cast<int>(fields.size());
108  spatial_dim_ = expectedColumns - 2;
109  MFEM_VERIFY(spatial_dim_ >= 1 && spatial_dim_ <= kMaxDim,
110  "TabulatedFunction: input coordinates not well read in CSV file. Please check "
111  "data.");
112  } else if (static_cast<int>(fields.size()) != expectedColumns) {
113  throw std::runtime_error("TabulatedFunction: line " + std::to_string(lineNo) + " contains " +
114  std::to_string(fields.size()) + " columns, " +
115  std::to_string(expectedColumns) + " expected");
116  }
117 
118  Sample s;
119  s.t = std::stod(fields[0]);
120  for (int d = 0; d < spatial_dim_; ++d) s.x[d] = std::stod(fields[1 + d]);
121  s.F = std::stod(fields[1 + spatial_dim_]);
122  samples.push_back(s);
123  }
124  MFEM_VERIFY(!samples.empty(), "TabulatedFunction: CSV file '"
125  << path.string()
126  << "' contains no data line (only a header, or empty file).");
127  return samples;
128 }

174 void TabulatedFunction::build_grid(std::vector<Sample>& samples) {
175  std::vector<double> tRaw;
176  tRaw.reserve(samples.size());
177  std::array<std::vector<double>, kMaxDim> xRaw;
178  for (int d = 0; d < spatial_dim_; ++d) {
179  xRaw[d].reserve(samples.size());
180  }
181 
182  for (const auto& s : samples) {
183  tRaw.push_back(s.t);
184  for (int d = 0; d < spatial_dim_; ++d) {
185  xRaw[d].push_back(s.x[d]);
186  }
187  }
188 
189  t_axis_ = unique_sorted(tRaw);
190  for (int d = 0; d < spatial_dim_; ++d) {
191  x_axis_[d] = unique_sorted(xRaw[d]);
192  }
193 
194  std::size_t expected = t_axis_.size();
195  for (int d = 0; d < spatial_dim_; ++d) expected *= x_axis_[d].size();
196  if (expected != samples.size()) {
197  throw std::runtime_error("TabulatedFunction: CSV file is not a structured grid (expected " +
198  std::to_string(expected) + ", received " +
199  std::to_string(samples.size()) + ")");
200  }
201 
202  auto is_uniform = [](const std::vector<double>& axis) {
203  if (axis.size() < 3) return true;
204  const double h = axis[1] - axis[0];
205  const double tol = 1e-9 * std::max(1.0, std::abs(h));
206  for (std::size_t i = 2; i < axis.size(); ++i) {
207  if (std::abs((axis[i] - axis[i - 1]) - h) > tol) return false;
208  }
209  return true;
210  };
211  t_uniform_ = is_uniform(t_axis_);
212  for (int d = 0; d < spatial_dim_; ++d) x_uniform_[d] = is_uniform(x_axis_[d]);
213 
214  data_.resize(expected);
215 
216  auto idx_of = [](const std::vector<double>& axis, double val) {
217  return static_cast<int>(std::lower_bound(axis.begin(), axis.end(), val) - axis.begin());
218  };
219 
220  std::vector<bool> filled(expected, false);
221 
222  for (const auto& s : samples) {
223  const int it = idx_of(t_axis_, s.t);
224  std::array<int, kMaxDim> ix{};
225  for (int d = 0; d < spatial_dim_; ++d) ix[d] = idx_of(x_axis_[d], s.x[d]);
226  const std::size_t k = flat_index(it, ix);
227  if (filled[k]) {
228  std::ostringstream oss;
229  oss << "TabulatedFunction: duplicate grid point in CSV file (t = " << s.t;
230  for (int d = 0; d < spatial_dim_; ++d) oss << ", x" << d + 1 << " = " << s.x[d];
231  oss << ")";
232  throw std::runtime_error(oss.str());
233  }
234  filled[k] = true;
235  data_[k] = s.F;
236  }
237 }

295 std::pair<int, double> TabulatedFunction::locate(const std::vector<double>& axis, bool uniform,
296  double val) const {
297  const int n = static_cast<int>(axis.size());
298  if (n == 1) return {0, 0.0};
299 
300  if (uniform) {
301  const double h = axis[1] - axis[0];
302  const int i = std::clamp(static_cast<int>(std::floor((val - axis[0]) / h)), 0, n - 2);
303  return {i, std::clamp((val - axis[i]) / h, 0.0, 1.0)};
304  }
305  int i = static_cast<int>(std::upper_bound(axis.begin(), axis.end(), val) - axis.begin()) - 1;
306  i = std::clamp(i, 0, n - 2);
307  return {i, std::clamp((val - axis[i]) / (axis[i + 1] - axis[i]), 0.0, 1.0)};
308 }

335 double TabulatedFunction::interpolate(double t, std::span<const double> pos) const {
336  MFEM_VERIFY(static_cast<int>(pos.size()) == spatial_dim_,
337  "TabulatedFunction: point dimension not compatible with the current table");
338 
339  auto [it, at] = locate(t_axis_, t_uniform_, t);
340 
341  std::array<int, kMaxDim> ix{};
342  std::array<double, kMaxDim> ax{};
343  for (int d = 0; d < spatial_dim_; ++d) {
344  auto [i, a] = locate(x_axis_[d], x_uniform_[d], pos[d]);
345  ix[d] = i;
346  ax[d] = a;
347  }
348 
349  auto local_lerp = [](double a, double b, double w) { return a + w * (b - a); };
350 
351  auto spatialAt = [&](int itc) {
352  const int nCorners = 1 << spatial_dim_;
353  double acc = 0.0;
354  for (int c = 0; c < nCorners; ++c) {
355  double w = 1.0;
356  bool valid = true;
357  std::array<int, kMaxDim> corner = ix;
358  for (int d = 0; d < spatial_dim_; ++d) {
359  const bool hi = (c >> d) & 1;
360  // No upper neighbour on a single-point axis
361  if (hi && x_axis_[d].size() == 1) {
362  valid = false;
363  break;
364  }
365  corner[d] += hi ? 1 : 0;
366  w *= hi ? ax[d] : (1.0 - ax[d]);
367  }
368  if (valid) acc += w * data_[flat_index(itc, corner)];
369  }
370  return acc;
371  };
372 
373  // Stationary field (single time value): no time interpolation
374  if (t_axis_.size() == 1) return spatialAt(0);
375  return local_lerp(spatialAt(it), spatialAt(it + 1), at);
376 }
