216  template <typename Solv, typename Prec>
217  inline void operator()(Solv&& solv, Prec&& prec) {
218  using TT = std::decay_t<decltype(solv)>;
219  using PP = std::decay_t<decltype(prec)>;
220  UtilsSolvers::print_solver_name(solv);
221  UtilsSolvers::print_prec_name(prec);
222  if constexpr (!std::is_same_v<PP, sptr<std::monostate>>) {
223  if constexpr (is_in_variant_v<TT, VIterativeSolver>) {
224  MFEM_VERIFY((is_in_variant_v<PP, VIterativePrecond>),
225  "SetPrecondSolver: IterativeSolver objects can only be associated with an "
226  "IterativePreconditionner objects");
227  if constexpr (is_in_variant_v<PP, VIterativePrecond>) {
228  SlothInfo::debug("SetPrecondSolver: setting iterative preconditionner");
229  UtilsSolvers::setter_mfem(solv, prec);
230  }
231  }
232  if constexpr (is_in_variant_v<TT, VHypreSolver>) {
233  MFEM_VERIFY((is_in_variant_v<PP, VHyprePrecond>),
234  "SetPrecondSolver: HypreSolver objects can only be associated with an "
235  "HyprePreconditionner objects");
236  if constexpr (is_in_variant_v<PP, VHyprePrecond>) {
237  SlothInfo::debug("SetPrecondSolver: setting hypre preconditionner (not hypre smoother)");
238  UtilsSolvers::setter_hypre(solv, prec);
239  }
240  }
241 #ifdef SLOTH_USE_MUMPS
242  if constexpr (is_in_variant_v<TT, VDirectSolver>) {
243  MFEM_VERIFY(false,
244  "SetPrecondSolver: direct solvers (MUMPS) do not use a preconditioner; "
245  "use PreconditionerType::NO or HyprePreconditionerType::NO");
246  }
247 #endif
248  }
249 
250  if constexpr (!std::is_same_v<TT, sptr<std::monostate>>) {
251  SlothInfo::debug("SetPrecondSolver: setting operator ");
252  solv->SetOperator(this->op);
253  }
254  }
