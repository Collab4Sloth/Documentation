62 /**
63  * @brief Refine the mesh once, based solely on the error estimated on the
64  * pilot variable (`var_id_`).
65  *
66  * @details Builds a fresh estimator via `amr_estimator_->get_value()`,
67  * configures a `mfem::ThresholdRefiner` with a purely local
68  * error goal (`SetTotalErrorFraction(0.0)`,
69  * `SetLocalErrorGoal(amr_max_elem_error_)`) and conforming
70  * refinement preferred where possible
71  * (`PreferConformingRefinement()`), subject to `amr_nc_limit_`.
72  * Candidates are computed via `MarkWithoutRefining()` (without
73  * mutating the mesh), filtered by `FilterRefinedCandidates()`
74  * (max refinement depth), and the decision to mutate is based
75  * on a global reduction (`mesh.ReduceInt()`) rather than the
76  * local candidate count, to stay correct under MPI when some
77  * ranks have no local candidate. If the mesh was refined,
78  * `amr_estimator_->UpdateFluxSpaces()` is called to
79  * resynchronize the estimator's internal flux spaces before it
80  * is destroyed.
81  *
82  * @tparam VAR Variable container type this AMR strategy operates on.
83  * @param vars Collection of variables sharing the mesh managed by this
84  * AMR object; only `vars[var_id_]` is consulted.
85 
86  * @return true The mesh was refined.
87  * @return false No element exceeded the local error goal; the mesh is
88  * unchanged.
89  */
90 template <class VAR>
91 bool SingleVariableAMR<VAR>::Refine(VAR& vars) {
92  this->EnsureNCMeshIfNeeded();
93  this->EnsureCriteriaSet();
94  auto& var = vars[this->var_id_];
95  auto estimator =
96  this->amr_estimator_->get_value(var.get_ref_gf(), *var.get_fespace(), this->par_mesh_);
97 
98  mfem::ThresholdRefiner refiner(*estimator);
99  refiner.SetTotalErrorFraction(0.0);
100  refiner.SetLocalErrorGoal(this->amr_max_elem_error_);
101  refiner.PreferConformingRefinement();
102  refiner.SetNCLimit(this->amr_nc_limit_);
103 
104  mfem::Array<mfem::Refinement> refinements;
105  refiner.MarkWithoutRefining(this->par_mesh_, refinements);
106  refinements = this->FilterRefinedCandidates(refinements);
107  int num_marked = this->par_mesh_.ReduceInt(refinements.Size());
108 
109  bool did_refine = (num_marked != 0);
110  if (did_refine) {
111  this->par_mesh_.GeneralRefinement(refinements, -1, this->amr_nc_limit_);
112  this->amr_estimator_->UpdateFluxSpaces();
113  }
114 
115  return did_refine;
116 }

118 /**
119  * @brief Derefine the mesh once, based solely on the error estimated on
120  * the pilot variable (`var_id_`).
121  *
122  * @details Builds a fresh estimator via `amr_estimator_->get_value()`,
123  * then delegates both error estimation and mesh mutation to
124  * `mfem::ThresholdDerefiner::Apply()`, with a threshold scaled
125  * down (`amr_scale_down_factor_ *`) relative to `amr_max_elem_error_` used for
126  * refinement, to provide hysteresis and avoid oscillating
127  * refine/derefine cycles on the same elements. As in `Refine()`,
128  * if the mesh was derefined, `amr_estimator_->UpdateFluxSpaces()`
129  * is called to resynchronize the estimator's internal flux
130  * spaces before it is destroyed.
131  *
132  * @tparam VAR Variable container type this AMR strategy operates on.
133  * @param vars Collection of variables sharing the mesh managed by this
134  * AMR object; only `vars[var_id_]` is consulted.
135  * @param auxvars Ignored by this strategy (logs a warning if non-empty);
136  * use `MultiVariableMaxAMR` if auxiliary variables must
137  * influence the refinement decision.
138  * @return true The mesh was derefined.
139  * @return false No element met the derefinement threshold; the mesh is
140  * unchanged.
141  */
142 template <class VAR>
143 bool SingleVariableAMR<VAR>::Derefine(VAR& vars, std::vector<VAR*> auxvars) {
144  if (!auxvars.empty()) {
145  SlothInfo::verbose("SingleVariableAMR::Derefine",
146  " auxiliary variables are ignored; use MultiVariableMaxAMR if they must "
147  "influence refinement.");
148  }
149  this->EnsureNCMeshIfNeeded();
150  this->EnsureCriteriaSet();
151  auto& var = vars[this->var_id_];
152  auto estimator =
153  this->amr_estimator_->get_value(var.get_ref_gf(), *var.get_fespace(), this->par_mesh_);
154  mfem::ThresholdDerefiner derefiner(*estimator);
155  derefiner.SetThreshold(this->amr_scale_down_factor_ * this->amr_max_elem_error_);
156  derefiner.SetNCLimit(this->amr_nc_limit_);
157 
158  bool did_derefine = derefiner.Apply(this->par_mesh_);
159 
160  if (did_derefine) {
161  this->amr_estimator_->UpdateFluxSpaces();
162  }
163 
164  return did_derefine;
165 }
