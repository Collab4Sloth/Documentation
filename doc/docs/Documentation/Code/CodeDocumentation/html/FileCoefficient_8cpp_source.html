165 double FileCoefficient::value(double time, std::span<const double> coordinates) const {
166  if (this->geometry_ == Geometry::Cartesian) {
167  return this->prefactor_ * this->field_->interpolate(time, coordinates);
168  }
169 
170  const int n = this->field_->input_dim();
171  MFEM_VERIFY(static_cast<int>(coordinates.size()) >= n + 1,
172  "FileCoefficient: mesh dimension too small for this axisymmetric field ("
173  << coordinates.size() << " coordinates, CSV input dimension = " << n << ")");
174  std::array<double, 2> q{
175  std::hypot(coordinates[0] - this->center_[0], coordinates[1] - this->center_[1]), 0.0};
176  if (n == 2) q[1] = coordinates[2];
177  return this->prefactor_ * this->field_->interpolate(time, std::span<const double>(q.data(), n));
178 }
