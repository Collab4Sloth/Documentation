55 /**
56  * @brief Refine the shared mesh once, based on the per-element maximum
57  * error across all variables in `vars`.
58  *
59  * @details For each variable, builds a fresh estimator via
60  * `amr_estimator_->get_value()` and reads its per-element local
61  * errors (this only estimates the error; it does not mutate the
62  * mesh). These are combined element-wise into `combined_error`
63  * via `max()`, so that an element is a refinement candidate as
64  * soon as ANY variable needs it resolved there. Candidates are
65  * filtered by `FilterRefinedCandidates()` (max refinement
66  * depth), and the decision to mutate is based on a global
67  * reduction (`mesh.ReduceInt()`) rather than the local
68  * candidate count, to stay correct under MPI when some ranks
69  * have no local candidate. The mesh is then mutated exactly
70  * once, via `GeneralRefinement()` — unlike calling
71  * `mfem::ThresholdRefiner::Apply()` separately per variable,
72  * which would mutate the mesh once per variable and require
73  * resynchronizing every other variable's finite element space
74  * in between.
75  *
76  * @tparam VAR Variable container type this AMR strategy operates on.
77  * @param vars Collection of variables sharing the mesh managed by this
78  * AMR object.
79  * @return true At least one element was marked and the mesh was refined.
80  * @return false No element exceeded the error threshold; the mesh is
81  * unchanged.
82  */
83 template <class VAR>
84 bool MultiVariableMaxAMR<VAR>::Refine(VAR& vars) {
85  this->EnsureNCMeshIfNeeded();
86  this->EnsureCriteriaSet();
87  const int ne = this->par_mesh_.GetNE();
88  mfem::Vector combined_error(ne);
89  combined_error = 0.0;
90 
91  auto accumulate_error = [&](auto& var) {
92  auto estimator =
93  this->amr_estimator_->get_value(var.get_ref_gf(), *var.get_fespace(), this->par_mesh_);
94  const mfem::Vector& local_errors = estimator->GetLocalErrors();
95  for (int e = 0; e < ne; e++) {
96  combined_error(e) = std::max(combined_error(e), local_errors(e));
97  }
98  };
99 
100  for (unsigned int iv = 0; iv < vars.get_variables_number(); iv++) {
101  accumulate_error(vars[iv]);
102  }
103 
104  mfem::Array<mfem::Refinement> refinements;
105  for (int e = 0; e < ne; e++) {
106  if (combined_error(e) > this->amr_max_elem_error_) {
107  refinements.Append(mfem::Refinement(e));
108  }
109  }
110 
111  refinements = this->FilterRefinedCandidates(refinements);
112  int num_marked = this->par_mesh_.ReduceInt(refinements.Size());
113  bool did_refine = (num_marked != 0);
114 
115  if (did_refine) {
116  this->par_mesh_.GeneralRefinement(refinements, -1, this->amr_nc_limit_);
117  }
118 
119  return did_refine;
120 }

122 /**
123  * @brief Derefine the shared mesh once, based on the per-element maximum
124  * error across all variables in `vars`.
125  *
126  * @details Mirrors `Refine()`: for each variable, builds a fresh
127  * estimator via `amr_estimator_->get_value()`, and the
128  * per-element error is combined via `max()` across all
129  * variables, so that an element is a derefinement candidate
130  * only if its error stays low for EVERY variable (a
131  * conservative criterion, since derefining would otherwise risk
132  * losing resolution needed by a variable that did not
133  * contribute to the decision). The combined error vector is
134  * then passed once to `mfem::ParMesh::DerefineByError()`, which
135  * internally aggregates by SUM the errors of sibling elements
136  * sharing the same parent before comparing to
137  * `derefine_threshold` — hence the threshold is scaled down
138  * (`this->amr_scale_down_factor_ *`) relative to `amr_max_elem_error_` used for
139  * refinement, and may need further empirical recalibration
140  * (e.g. multiplied by the number of siblings) depending on the
141  * observed derefinement behavior.
142  *
143  * @tparam VAR Variable container type this AMR strategy operates on.
144  * @param vars Collection of variables sharing the mesh managed by this
145  * AMR object.
146  * @param auxvars Auxiliary variable collections whose error also
147  * contributes to `combined_error`, if any (empty by
148  * default). Only taken into account by the caller when
149  * `AMRBase::SetIncludeAuxVars(true)` was set.
150  * @return true At least one group of elements was derefined.
151  * @return false No group of elements met the derefinement criterion; the
152  * mesh is unchanged.
153  */
154 template <class VAR>
155 bool MultiVariableMaxAMR<VAR>::Derefine(VAR& vars, std::vector<VAR*> auxvars) {
156  this->EnsureNCMeshIfNeeded();
157  this->EnsureCriteriaSet();
158 
159  const int ne = this->par_mesh_.GetNE();
160  mfem::Vector combined_error(ne);
161  combined_error = 0.0;
162 
163  auto accumulate_error = [&](auto& var) {
164  auto estimator =
165  this->amr_estimator_->get_value(var.get_ref_gf(), *var.get_fespace(), this->par_mesh_);
166  const mfem::Vector& local_errors = estimator->GetLocalErrors();
167  for (int e = 0; e < ne; e++) {
168  combined_error(e) = std::max(combined_error(e), local_errors(e));
169  }
170  };
171  for (unsigned int iv = 0; iv < vars.get_variables_number(); iv++) {
172  accumulate_error(vars[iv]);
173  }
174  for (auto* auxvar : auxvars) {
175  for (std::size_t i = 0; i < auxvar->get_variables_number(); i++) {
176  accumulate_error((*auxvar)[i]);
177  }
178  }
179 
180  const double derefine_threshold = this->amr_scale_down_factor_ * this->amr_max_elem_error_;
181 
182  bool did_derefine =
183  this->par_mesh_.DerefineByError(combined_error, derefine_threshold, this->amr_nc_limit_);
184 
185  return did_derefine;
186 }
