72  // Phase 1: build safely. Each element is owned by a std::unique_ptr in
73  // `storage`, so std::vector reallocation (if capacity were ever exceeded)
74  // would only move unique_ptr's themselves, never touch the pointee -
75  // sidesteps SpatialDiscretization having no move constructor entirely.
76  std::vector<std::unique_ptr<SpatialDiscretization<T, DIM>>> storage;
77  storage.reserve(N);
78 
79  // fe_order is always the 2nd parameter of every SpatialDiscretization
80  // constructor - grab it before args... is consumed by forwarding.
81  const int fe_order = std::get<1>(std::forward_as_tuple(args...));
82 
83  // storage[0]: built by forwarding args... to whichever constructor matches.
84  storage.push_back(std::make_unique<SpatialDiscretization<T, DIM>>(std::forward<Args>(args)...));
85 
86  // storage[1..N-1]: share storage[0]'s mesh, own their own FE space.
87  mfem::ParMesh* shared_mesh = storage.front()->get_mesh();
88  for (std::size_t i = 1; i < N; ++i) {
89  storage.push_back(
90  std::make_unique<SpatialDiscretization<T, DIM>>(shared_mesh, fe_order, shared_is_periodic));
91  }
92 
93  // Phase 2: transfer ownership to the caller as raw pointers (SPAS). Each
94  // unique_ptr::release() hands off ownership without deleting anything;
95  // `storage` itself is emptied of ownership responsibility as it goes out
96  // of scope right after (its destructor runs on now-null unique_ptr's).
97  std::vector<SpatialDiscretization<T, DIM>*> result;
98  result.reserve(N);
99  for (auto& ptr : storage) {
100  result.push_back(ptr.release());
101  }
